
interface Notificacao {
    void enviar(String mensagem);
}

class NotificacaoEmail implements Notificacao {
    public void enviar(String mensagem) {
        System.out.println("E-mail enviado: " + mensagem);
    }
}

class NotificacaoSMS implements Notificacao {
    public void enviar(String mensagem) {
        System.out.println("SMS enviado: " + mensagem);
    }
}

class FabricaNotificacao {
    public static Notificacao criar(String tipo) {
        if (tipo.equalsIgnoreCase("email")) {
            return new NotificacaoEmail();
        }

        if (tipo.equalsIgnoreCase("sms")) {
            return new NotificacaoSMS();
        }

        throw new IllegalArgumentException(
            "Tipo de notificacao invalido"
        );
    }
}

interface EstrategiaNotificacao {
    void enviar(String mensagem);
}

class EstrategiaEmail implements EstrategiaNotificacao {
    public void enviar(String mensagem) {
        System.out.println("E-mail enviado: " + mensagem);
    }
}

class EstrategiaSMS implements EstrategiaNotificacao {
    public void enviar(String mensagem) {
        System.out.println("SMS enviado: " + mensagem);
    }
}

class ServicoNotificacao {
    private EstrategiaNotificacao estrategia;

    public ServicoNotificacao(
        EstrategiaNotificacao estrategia
    ) {
        this.estrategia = estrategia;
    }

    public void definirEstrategia(
        EstrategiaNotificacao estrategia
    ) {
        this.estrategia = estrategia;
    }

    public void enviar(String mensagem) {
        estrategia.enviar(mensagem);
    }
}

public class Main {
    public static void main(String[] args) {
        System.out.println("=== FACTORY ===");

        Notificacao email =
            FabricaNotificacao.criar("email");
        email.enviar("Projeto da DIO iniciado!");

        Notificacao sms =
            FabricaNotificacao.criar("sms");
        sms.enviar("Projeto executado com sucesso!");

        System.out.println("=== STRATEGY ===");

        ServicoNotificacao servico =
            new ServicoNotificacao(new EstrategiaEmail());

        servico.enviar("Enviando por e-mail.");

        servico.definirEstrategia(new EstrategiaSMS());
        servico.enviar("Enviando por SMS.");
    }
}
